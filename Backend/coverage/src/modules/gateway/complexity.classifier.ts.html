import { GroqService } from "../providers/groq/groq.service";
import { CLASSIFIER_TIER_CONFIG } from "./gateway.config";
import type { ComplexityTier } from "./gateway.types";
 
export async function classifyComplexity(prompt: string): Promise<ComplexityTier> {
  const classifier = new GroqService({ defaultModel: CLASSIFIER_TIER_CONFIG.model });
 
  const result = await classifier.generate({
    prompt: `Classify the complexity of this user request as exactly one word - "simple", "medium", or "complex". Only output that one word.\n\nRequest: """${prompt}"""`,
    maxOutputTokens: 5,
  });
 
  const answer = result.response.trim().toLowerCase();
  if (answer.includes("simple")) {
    return "simple";
  }
  if (answer.includes("complex")) {
    return "complex";
  }
 
  return "medium";
}
